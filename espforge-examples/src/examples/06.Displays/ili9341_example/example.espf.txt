==================
ILI9341 TFT EXAMPLE
==================

# espforge
  <project name="ili9341_example" />
  <chip type="esp32c3" />
  <runtime type="blocking" />

# peripherals
  <spi id="spi2" sclk="3" mosi="4" frequency="10MHz" /> // MISO is not connected for ili9341
  <gpio id="pin_dc" pin="6" direction="output" />
  <gpio id="pin_rst" pin="7" direction="output" />
  <gpio id="pin_cs" pin="5" direction="output" />

# components
  <spi_component id="main_spi" spi="$spi2" />

# devices
  <ili9341 id="display" spi_component="$main_spi"
           dc="$pin_dc" rst="$pin_rst" cs="$pin_cs" />
